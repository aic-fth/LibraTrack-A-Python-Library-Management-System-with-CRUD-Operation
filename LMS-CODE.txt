# ─────────────────────────────────────────────
#  STATE
# ─────────────────────────────────────────────
books       = {}
next_id     = [1]
_after      = [None]
_mode       = ["add"]

# ─────────────────────────────────────────────
#  HELPERS
# ─────────────────────────────────────────────
def gen_id():
    bid = f"BK-{next_id[0]:04d}"
    next_id[0] += 1
    return bid

def toast(msg, color=GREEN):
    toast_label.config(text=msg, fg=color)
    toast_bar.config(bg=color)
    if _after[0]:
        root.after_cancel(_after[0])
    _after[0] = root.after(3200, lambda: (
        toast_label.config(text="  Ready", fg=TEXT_SUB),
        toast_bar.config(bg=BORDER)
    ))

def set_mode(mode):
    _mode[0] = mode
    if mode == "add":
        add_btn.config(text="  Add Book", bg=ROSE, fg="#FFFFFF",
                       activebackground=ROSE_LIGHT)
        add_btn.bind("<Enter>", lambda e: add_btn.config(bg=ROSE_LIGHT))
        add_btn.bind("<Leave>", lambda e: add_btn.config(bg=ROSE))
        mode_label.config(text="NEW BOOK", fg=ROSE, bg=ROSE_BG)
        id_display.config(text="ID will be assigned", fg=TEXT_DIM)
        clear_form_fields()
    else:
        add_btn.config(text="  Save Changes", bg=GREEN, fg="#FFFFFF",
                       activebackground="#82B894")
        add_btn.bind("<Enter>", lambda e: add_btn.config(bg="#82B894"))
        add_btn.bind("<Leave>", lambda e: add_btn.config(bg=GREEN))
        mode_label.config(text="EDITING", fg=GREEN, bg=GREEN_BG)

def refresh_table(select_id=None):
    book_table.delete(*book_table.get_children())
    for i, (bid, (t, a, g)) in enumerate(books.items()):
        tag = "odd" if i % 2 == 0 else "even"
        book_table.insert("", tk.END, iid=bid,
                          values=(bid, t, a, g), tags=(tag,))
    update_stats()
    if select_id and select_id in books:
        book_table.selection_set(select_id)
        book_table.see(select_id)

def update_stats():
    n = len(books)
    stat_num.config(text=str(n))
    stat_sub.config(text="book" if n == 1 else "books")

def clear_form_fields():
    for e in (ent_title, ent_author, ent_genre):
        e.delete(0, tk.END)
    search_var.set("")
    id_display.config(text="ID will be assigned", fg=TEXT_DIM)

def full_clear():
    clear_form_fields()
    book_table.selection_remove(*book_table.selection())
    set_mode("add")
    ent_title.focus()

def fill_form(bid):
    t, a, g = books[bid]
    ent_title.delete(0, tk.END);  ent_title.insert(0, t)
    ent_author.delete(0, tk.END); ent_author.insert(0, a)
    ent_genre.delete(0, tk.END);  ent_genre.insert(0, g)
    search_var.set(bid)
    id_display.config(text=bid, fg=ROSE)
    set_mode("edit")

def on_row_select(event=None):
    sel = book_table.selection()
    if sel:
        fill_form(sel[0])

# ─────────────────────────────────────────────
#  CRUD ACTIONS
# ─────────────────────────────────────────────
def action_add_or_save():
    t = ent_title.get().strip()
    a = ent_author.get().strip()
    g = ent_genre.get().strip()
    if not t or not a or not g:
        toast("  Fill in all three fields.", RED)
        return
    if _mode[0] == "add":
        bid = gen_id()
        books[bid] = (t, a, g)
        refresh_table(bid)
        toast(f"  Added  {bid} — {t}", GREEN)
        full_clear()
    else:
        bid = search_var.get().strip().upper()
        if bid not in books:
            toast("  Book ID not found. Cannot save.", RED)
            return
        books[bid] = (t, a, g)
        refresh_table(bid)
        toast(f"  Saved  {bid} — {t}", GREEN)
        full_clear()

def action_delete():
    sel = book_table.selection()
    bid_field = search_var.get().strip().upper()
    bid = sel[0] if sel else (bid_field if bid_field in books else None)
    if not bid:
        toast("  Select a row or enter an ID to delete.", RED)
        return
    if bid not in books:
        toast(f"  {bid} not found.", RED)
        return
    title = books[bid][0]
    ok = messagebox.askyesno("Delete Book",
        f'Delete "{title}" ({bid})?\n\nThis cannot be undone.',
        icon="warning")
    if ok:
        del books[bid]
        refresh_table()
        toast(f"  Deleted  {bid} — {title}", ROSE)
        full_clear()

def action_search():
    q = search_var.get().strip().upper()
    if not q:
        toast("  Enter a Book ID to search.", RED)
        return
    if q in books:
        fill_form(q)
        book_table.selection_set(q)
        book_table.see(q)
        toast(f"  Found  {q} — {books[q][0]}", GREEN)
    else:
        toast(f"  No book with ID  {q}", RED)

def action_exit():
    if messagebox.askokcancel("Exit", "Close LibraTrack?"):
        root.destroy()

def btn(parent, text, cmd, bg=CARD2, fg=TEXT,
        hbg=BORDER, hfg=TEXT, font=UI_B,
        padx=16, pady=9, width=0, anchor="center"):
    b = tk.Button(parent, text=text, command=cmd,
                  bg=bg, fg=fg, activebackground=hbg, activeforeground=hfg,
                  font=font, relief="flat", bd=0, cursor="hand2",
                  padx=padx, pady=pady, width=width, anchor=anchor)
    b.bind("<Enter>", lambda e: b.config(bg=hbg, fg=hfg))
    b.bind("<Leave>", lambda e: b.config(bg=bg,  fg=fg))
    return b

def field(parent, label):
    wrap = tk.Frame(parent, bg=SURFACE)
    wrap.pack(fill=tk.X, pady=(0, 10))
    tk.Label(wrap, text=label, font=UI_SM, bg=SURFACE, fg=TEXT_SUB
             ).pack(anchor="w", pady=(0, 3))
    e = tk.Entry(wrap, font=UI, bg="#FFFFFF", fg=TEXT,
                 insertbackground=ROSE, relief="flat",
                 highlightthickness=1,
                 highlightbackground=BORDER,
                 highlightcolor=ROSE, bd=0)
    e.pack(fill=tk.X, ipady=7)
    return e

def divider(parent, pady=10):
    tk.Frame(parent, bg=BORDER, height=1).pack(fill=tk.X, pady=pady)

def caps_label(parent, text):
    tk.Label(parent, text=text, font=UI_CAPS,
             bg=SURFACE, fg=TEXT_DIM).pack(anchor="w", pady=(12, 4))
